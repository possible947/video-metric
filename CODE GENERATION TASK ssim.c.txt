double compute_ssim(
    const char *orig,
    const char *test,
    int threads) {
    char escaped_orig[PATH_MAX];
    char escaped_test[PATH_MAX];
    escape_path(orig, escaped_orig, sizeof(escaped_orig));
    escape_path(test, escaped_test, sizeof(escaped_test));

    char cmd[4096];
    snprintf(cmd, sizeof(cmd),
        "./ffmpeg -i %s -i %s -vf \"ssim\" -f null - -threads %d",
        escaped_orig, escaped_test, threads > 0 ? threads : 1);

    FILE *fp = popen(cmd, "r");
    if (!fp) { perror("popen"); return NAN; }

    double score = NAN;
    char line[1024];
    while (fgets(line, sizeof(line), fp)) {
        if (strncmp(line, "SSIM", 4) == 0) {
            char *pos = strstr(line, "All:");
            if (pos) {
                pos += 4;
                while (*pos && isspace((unsigned char)*pos)) pos++;
                score = strtod(pos, NULL);
            }
            break;
        }
    }

    pclose(fp);
    return score;
