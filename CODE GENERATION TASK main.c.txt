int main(int argc, char *argv[]) {
    struct cli_options opts;
    if (parse_cli(argc, argv, &opts) != 0) {
        print usage;
        return 1;
    }
    if (!opts.compute_ssim && !opts.compute_vmaf) {
        error("Specify at least one metric");
        return 1;
    }

    char status_buf[256];
    if (check_environment(status_buf, sizeof(status_buf)) != 0) {
        printf("%s\n", status_buf);
        return 1;
    }
    printf("%s\n", status_buf);

    if (!file_exists(opts.orig) || !file_exists(opts.test)) {
        error("Input files missing");
        return 1;
    }

    double ssim_val = NAN;
    if (opts.compute_ssim) {
        ssim_val = compute_ssim(opts.orig, opts.test, opts.num_threads);
        printf("SSIM: %.6f\n", ssim_val);
    }

    double vmaf_val = NAN;
    if (opts.compute_vmaf) {
        char json_path[PATH_MAX];
        create_temp_dir(json_path, sizeof(json_path));
        vmaf_val = compute_vmaf(opts.orig, opts.test, model_path(opts.resolution),
                                opts.num_threads, json_path, sizeof(json_path));
        printf("VMAF: %.6f\n", vmaf_val);
        remove_temp_dir(json_path);
    }

    return 0;
}
