double compute_vmaf(
    const char *orig,
    const char *test,
    const char *model_path,
    int threads,
    char *json_path,
    size_t json_bufsize) {

    // 1. Create temporary file for log
    char template[] = "/tmp/vmaf_XXXXXX.json";
    int fd = mkstemp(template);
    if (fd < 0) { error; return NAN; }
    close(fd);
    // copy full path to json_path
    strncpy(json_path, template, json_bufsize);

    // 2. Escape paths
    char esc_orig[PATH_MAX];
    char esc_test[PATH_MAX];
    escape_path(orig, esc_orig, sizeof(esc_orig));
    escape_path(test, esc_test, sizeof(esc_test));

    // 3. Build command
    char cmd[4096];
    snprintf(cmd, sizeof(cmd),
        "./ffmpeg -i %s -i %s -filter_complex \"[0:v][1:v]libvmaf=model=path=%s:n_threads=%d:log_path=%s\" -f null - -threads %d",
        esc_orig, esc_test, model_path, threads > 0 ? threads : 1, json_path, threads > 0 ? threads : 1);

    // 4. Execute and capture output
    FILE *fp = popen(cmd, "r");
    if (!fp) { error; unlink(json_path); return NAN; }
    double score = NAN;
    char line[1024];
    while (fgets(line, sizeof(line), fp)) {
        if (strncmp(line, "VMAF score:", 11) == 0) {
            char *ptr = line + 11;
            while (*ptr && isspace((unsigned char)*ptr)) ptr++;
            score = strtod(ptr, NULL);
            break;
        }
    }
    pclose(fp);

    // 5. Clean up log
    unlink(json_path);

    return score;
}
